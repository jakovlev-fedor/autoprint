# ══════════════════════════════════════════════════════════════
# РАЗВЁРТЫВАНИЕ AutoPrint
# ══════════════════════════════════════════════════════════════

# ── 1. Требования к системе ──────────────────────────────────
# Windows 10/11 x64
# Python 3.10+
# AutoCAD (любая версия с COM-интерфейсом, лицензированная)
# ODA File Converter 27.1.0 (или другая версия — скорректируйте ODA_PATH)
#   Скачать: https://www.opendesign.com/guestfiles/oda_file_converter

# ── 2. Установка Python-зависимостей ─────────────────────────
pip install customtkinter pywin32 ezdxf pypdf psutil pyinstaller

# ── 3. Настройка AutoCAD ─────────────────────────────────────
# В AutoCAD создать именованный параметр страницы "Стандарт":
#   Файл → Диспетчер параметров листа → Создать → "Стандарт"
#   Настроить: принтер, формат бумаги, масштаб
# Профиль AutoCAD должен называться "<<Профиль без имени>>"
# (или изменить ACAD_PROFILE в orchestrator.py)

# ── 4. Структура размещения ──────────────────────────────────
# Сетевая папка ядра (например J:\Soft\AutoCAD AutoPrint\core\):
#   orchestrator.py
#   module2_version_filter.py
#   module3_dsd_generator.py
#   module4_print_engine.py
#   module5_pdf_distributor.py

# Папка проекта:
#   /ИМЯ_ПРОЕКТА/
#   └── !_Печать/
#       ├── print.exe          ← скомпилированный раннер
#       └── config/
#           └── config.ini     ← CorePath указывает на папку ядра

# ── 5. Компиляция print.exe ──────────────────────────────────
pyinstaller --onefile --windowed --name print print_runner.py

# Результат: dist/print.exe
# Скопировать в /ИМЯ_ПРОЕКТА/!_Печать/

# ── 6. Первый запуск ─────────────────────────────────────────
# 1. Открыть print.exe
# 2. Проверить CorePath в верхней строке
# 3. При необходимости — изменить и нажать "Сохранить"
# 4. Таблица автоматически заполнится объектами проекта
# 5. Нажать "Pipeline (Авто)"

# ── 7. Обновление ядра без перекомпиляции ────────────────────
# Отредактировать .py файлы в CorePath
# Следующий запуск Pipeline автоматически загрузит новую версию
# print.exe перекомпилировать НЕ нужно

# ── 8. Корректировка ODA_PATH ────────────────────────────────
# В orchestrator.py найти:
#   ODA_PATH = r"C:\Program Files\ODA\ODAFileConverter 27.1.0\ODAFileConverter.exe"
# Заменить на актуальный путь установки ODA File Converter

pyinstaller --onefile --windowed --name print print_runner.py
